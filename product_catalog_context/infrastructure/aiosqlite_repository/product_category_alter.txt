import aiosqlite
from httpx import delete
from api.repository.aiosqlite.errors_mapper import map_errors
from api.repository.aiosqlite.repository import DbRepository
from api.schemas.product_category import ProductCategoryCreate, ProductCategoryHierarchy, ProductCategoryUpdate

class ProductCategoryRepository(DbRepository):
  def __init__(self, db, name='ProductCategory'):
    super().__init__(db, name)

  def _format_category_hierarchy(self, rows):
    external_categories_by_id = {}
    all_categories_by_id = {}
    for row in rows:
      row_dict = dict(row)
      category_id = row_dict['id']
      all_categories_by_id[category_id] = {"id": category_id, "name": row_dict['name'], 'subcategories': []}
      if row_dict['depth'] == 0:
        external_categories_by_id[category_id] = all_categories_by_id[category_id]
      else:
        parent_id = row_dict['parent_id']
        all_categories_by_id[parent_id]['subcategories'].append(all_categories_by_id[category_id])
    return external_categories_by_id

  async def get_by_supercategory(self, supercategory_id) -> ProductCategoryHierarchy | None:
    get_by_supertype_query = """
      SELECT * FROM ProductCategory AS pc
      JOIN ProductCategoryClosure AS pcc ON pc.id  = pcc.subcategory_id
      WHERE pcc.supercategory_id = ?
      ORDER BY pcc.depth
    """
    try:
      async with self.db.execute(get_by_supertype_query, [supercategory_id]) as cursor:
        rows = await cursor.fetchall()
    except Exception as e:
      raise map_errors(e) from e
    
    if not rows:
      return None
    
    hierarchy_dict = self._format_category_hierarchy(rows)
    root_hierarchy = list(hierarchy_dict.values())[0]
  
    return ProductCategoryHierarchy(**root_hierarchy)
  
  async def get_all(self) -> list[ProductCategoryHierarchy]:
    get_all_query =  """
      SELECT child.*, pcc.* FROM ProductCategoryClosure AS pcc
      JOIN ProductCategory AS root ON root.id  = pcc.supercategory_id
      JOIN ProductCategory AS child ON child.id = pcc.subcategory_id
      WHERE root.parent_id IS NULL
      ORDER BY pcc.depth
    """
    try:
      async with self.db.execute(get_all_query) as cursor:
        rows = await cursor.fetchall()
    except Exception as e:
      raise map_errors(e) from e
    #print([dict(r) for r in rows])
    
    if not rows:
      return []
    
    hierarchy_dict = self._format_category_hierarchy(rows)

    hierarchy_list = [ProductCategoryHierarchy(**hierarchy_tree) for hierarchy_tree in hierarchy_dict.values()]
    return hierarchy_list
  


  async def insert(self, product_category: ProductCategoryCreate, **filters):
    adopt_subcategories = filters.get('adopt_subcategories')
    category_id = filters.get('id')
    insert_product_category_query = f"""
    INSERT INTO ProductCategory ({'id,'if category_id is not None else ''} name, parent_id)
    VALUES ({'?,' if category_id is not None else ''} ?, ?)
    """
    insert_product_category_closure_query = """
      INSERT INTO ProductCategoryClosure (supercategory_id, subcategory_id, depth)
      VALUES (?, ?, 0)
    """
    update_product_category_query= """
      UPDATE ProductCategory AS pc
      SET parent_id = ?
      WHERE id IN (
        SELECT subcategory_id FROM ProductCategoryClosure
        WHERE  supercategory_id = ? AND depth = 1
      ) 
    """
    try:
      insert_product_category_args = (category_id, product_category.name, product_category.supercategory_id) if category_id is not None else (product_category.name, product_category.supercategory_id)
      cursor = await self.db.execute(insert_product_category_query, insert_product_category_args)
      product_category_id = cursor.lastrowid
      await self.db.execute(insert_product_category_closure_query, (product_category_id, product_category_id))
      await self._insert_links(product_category_id, product_category, adopt_subcategories)
      if product_category.subcategories_id or adopt_subcategories:
        await self.db.execute(update_product_category_query, (product_category_id, product_category_id))
    except Exception as e:
      raise map_errors(e) from e
    
  async def delete(self, product_category_id, **filters):
    with_subcategories = filters.get('with_subcategories')
    delete_with_subcategories_query = """
      DELETE FROM ProductCategory
      WHERE id IN (
        SELECT subcategory_id FROM ProductCategoryClosure 
        WHERE supercategory_id = ?
      )
    """
    update_links_query = """
      UPDATE ProductCategoryClosure
      SET depth = depth - 1
      WHERE subcategory_id IN (
        SELECT subcategory_id FROM ProductCategoryClosure
        JOIN ProductCategory ON supercategory_id = id
        WHERE parent_id = ? 
      ) 
      AND 
      supercategory_id IN (
        SELECT supercategory_id FROM ProductCategoryClosure
        WHERE subcategory_id = ? 
      ) 
    """
    delete_links_query = """
      DELETE FROM ProductCategoryClosure
      WHERE supercategory_id = ? 
    """
    update_parents = """
      UPDATE ProductCategory
      SET parent_id = (
        SELECT parent_id FROM ProductCategory
        WHERE id = ?
      )
      WHERE parent_id = ?
    """
    delete_category = """
      DELETE FROM ProductCategory
      WHERE id = ?
    """
    cursor = None
    if with_subcategories:
      try:
        cursor = await self.db.execute(delete_with_subcategories_query, (product_category_id,))
        return cursor.rowcount
      except Exception as e:
        raise map_errors(e) from e
      finally:
        if cursor is not None:
          await cursor.close()
    else:
      try:
        
        cursor = await self.db.execute(delete_links_query, (product_category_id,))

        cursor = await self.db.execute(update_links_query, (product_category_id, product_category_id))
        cursor = await self.db.execute(update_parents, (product_category_id, product_category_id))

        cursor = await self.db.execute(delete_category, (product_category_id,))
        return cursor.rowcount
      except Exception as e:
        raise map_errors(e) from e
      finally:
        if cursor is not None:
          await cursor.close()
    
  async def update(self, product_category_id, product_category: ProductCategoryUpdate, **options):
    adopt_subcategories = options.get("adopt_subcategories")
 
    if adopt_subcategories:
      return await self._move_with_all_subcategories(product_category_id, product_category.supercategory_id)
    else:
      await self.delete(product_category_id)
      await self.insert(product_category, id=product_category_id)
      return 1
      

  async def _move_with_all_subcategories(self, product_category_id, supercategory_id):
    delete_links = """
        DELETE FROM ProductCategoryClosure
        WHERE subcategory_id IN (
            SELECT subcategory_id FROM ProductCategoryClosure
            WHERE supercategory_id = ?
          ) 
          AND 
          supercategory_id IN (
            SELECT supercategory_id FROM ProductCategoryClosure
            WHERE subcategory_id = ? AND supercategory_id != subcategory_id
          )
      """
    insert_all_links = """
      INSERT INTO ProductCategoryClosure (supercategory_id, subcategory_id, depth)
      SELECT 
        ancestors.supercategory_id,descendants.subcategory_id, descendants.depth+ 1+ ancestors.depth
      FROM (
        SELECT supercategory_id, subcategory_id, depth
        FROM ProductCategoryClosure
        WHERE subcategory_id = ?
      ) AS ancestors
      CROSS JOIN (
        SELECT supercategory_id, subcategory_id, depth
        FROM ProductCategoryClosure
        WHERE supercategory_id = ?
      ) AS descendants;
      """
    update_parent = """
      UPDATE ProductCategory
      SET parent_id = ?
      WHERE id = ?
      """
    cursor = await self.db.execute(delete_links, (product_category_id, product_category_id))
    cursor = await self.db.execute(insert_all_links, (supercategory_id, product_category_id))
    cursor = await self.db.execute(update_parent, (supercategory_id, product_category_id))
    return cursor.rowcount
  

  async def _insert_links_to_ancestors(self, product_category_id, product_category: ProductCategoryCreate):
    insert_to_parent_links_query = """
      INSERT INTO ProductCategoryClosure (supercategory_id, subcategory_id, depth)
      SELECT supercategory_id, ?, depth + 1 FROM ProductCategoryClosure
      WHERE subcategory_id = ?
    """
    await self.db.execute(insert_to_parent_links_query, (product_category_id, product_category.supercategory_id)) 

  async def _insert_links_to_specific_descendants(self, product_category_id, product_category: ProductCategoryCreate):
    conditions_list = ['supercategory_id = ?' for _ in product_category.subcategories_id]
    condition = ' OR '.join(conditions_list)
    insert_links_to_specific_descendants_query = f"""
      INSERT INTO ProductCategoryClosure (supercategory_id, subcategory_id, depth)
      SELECT ?, subcategory_id, depth + 1  FROM ProductCategoryClosure
      WHERE {condition}
    """
    await self.db.execute(insert_links_to_specific_descendants_query, (product_category_id, *product_category.subcategories_id))

  async def _insert_links_to_all_descendants(self, product_category_id, product_category: ProductCategoryCreate):
    insert_links_to_all_descendants_query = """
      INSERT INTO ProductCategoryClosure (supercategory_id, subcategory_id, depth)
      SELECT ?, subcategory_id, depth - 1 FROM ProductCategoryClosure
      WHERE supercategory_id = ? AND subcategory_id != ? AND depth != 0
    """
    await self.db.execute(insert_links_to_all_descendants_query, (product_category_id, product_category.supercategory_id, product_category_id))
  

  async def _update_links(self, product_category_id):
    update_links_query = """
      UPDATE ProductCategoryClosure
      SET depth = depth + 1
      WHERE subcategory_id IN (
        SELECT subcategory_id FROM ProductCategoryClosure WHERE supercategory_id = ? AND depth != 0
      ) AND supercategory_id IN (
        SELECT supercategory_id FROM ProductCategoryClosure WHERE subcategory_id = ? AND depth != 0
      )
    """
    await self.db.execute(update_links_query, (product_category_id, product_category_id))

  async def _update_links_to_specific_descendants(self, product_category: ProductCategoryCreate):
    conditions_list = ['supercategory_id = ?' for _ in product_category.subcategories_id]
    condition = ' OR '.join(conditions_list)
    
    update_links_query = f"""
      UPDATE ProductCategoryClosure
      SET depth = depth + 1000000
      WHERE subcategory_id IN (
        SELECT subcategory_id FROM ProductCategoryClosure WHERE {condition}
      ) AND supercategory_id IN (
        SELECT supercategory_id FROM ProductCategoryClosure WHERE subcategory_id = ?
      );
    """
    normalize_depth = """
      UPDATE ProductCategoryClosure
      SET depth = depth - 999999
      WHERE depth >= 1000000;
    """
    await self.db.execute(update_links_query, (*product_category.subcategories_id, product_category.supercategory_id))
    await self.db.execute(normalize_depth)


  async def _update_links_to_all_descendants(self, product_category_id, product_category: ProductCategoryCreate):
    
    update_links_query = """
      UPDATE ProductCategoryClosure
      SET depth = depth + 1000000
      WHERE subcategory_id IN (
        SELECT subcategory_id FROM ProductCategoryClosure WHERE supercategory_id = ? AND supercategory_id != subcategory_id
      ) AND supercategory_id IN (
        SELECT supercategory_id FROM ProductCategoryClosure WHERE subcategory_id = ? 
      ) AND subcategory_id != ?;
    """
    normalize_depth = """
      UPDATE ProductCategoryClosure
      SET depth = depth - 999999
      WHERE depth >= 1000000;
    """
    await self.db.execute(update_links_query, (product_category.supercategory_id, product_category.supercategory_id, product_category_id))
    await self.db.execute(normalize_depth)

  async def _insert_links(self, product_category_id, product_category: ProductCategoryCreate, adopt_subcategories=False):
    await self._insert_links_to_ancestors(product_category_id, product_category)
    if adopt_subcategories:
      cursor = await self.db.execute("SELECT * FROM ProductCategoryClosure")
      closure = await cursor.fetchall()
      print([dict(c) for c in closure])
      await self._update_links_to_all_descendants(product_category_id, product_category)
      cursor = await self.db.execute("SELECT * FROM ProductCategoryClosure")
      closure = await cursor.fetchall()
      print([dict(c) for c in closure])
      
      await self._insert_links_to_all_descendants(product_category_id, product_category)
      cursor = await self.db.execute("SELECT * FROM ProductCategoryClosure")
      closure = await cursor.fetchall()
      print([dict(c) for c in closure])
    elif product_category.subcategories_id is not None:

      
      await self._update_links_to_specific_descendants(product_category)


      await self._insert_links_to_specific_descendants(product_category_id, product_category)
